DECLARE n int;
BEGIN

CREATE OR REPLACE FUNCTION public.fn_auth_perm_class(p_code text) RETURNS text LANGUAGE sql IMMUTABLE PARALLEL SAFE SET search_path = pg_catalog, public AS $$
  SELECT CASE
    WHEN p_code = 'access.approve' THEN 'APPROVE'
    WHEN p_code IN ('employee.manage', 'role.manage', 'auth.credential.manage', 'auth.reset.request', 'access.request') THEN 'REQUEST'
    WHEN p_code ~ '^(customer|contact|activity|interaction|approval)\.' THEN 'BIZ'
    ELSE 'OTHER' END $$;
COMMENT ON FUNCTION public.fn_auth_perm_class(text) IS 'B1.3: phan loai quyen APPROVE/REQUEST/BIZ/OTHER cho SoD. Doi = migration moi.';
DELETE FROM public._prisma_migrations WHERE migration_name = '20261118000000_sales_perm_class';

DO $$ BEGIN
  IF public.fn_auth_perm_class('opportunity.read') <> 'OTHER' OR public.fn_auth_perm_class('customer.read') <> 'BIZ'
     OR public.fn_auth_perm_class('access.approve') <> 'APPROVE' OR public.fn_auth_perm_class('role.manage') <> 'REQUEST'
     OR EXISTS (SELECT 1 FROM public._prisma_migrations WHERE migration_name = '20261118000000_sales_perm_class')
     OR EXISTS (SELECT 1 FROM public.permission WHERE code ~ '^opportunity\.') THEN
    RAISE EXCEPTION 'Rollback Sales S5 (perm_class) chua sach.';
  END IF;
END $$;
COMMIT;
