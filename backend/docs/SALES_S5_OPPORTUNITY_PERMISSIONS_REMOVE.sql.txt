-- SISCOM CRM - Sales S5: ROLLBACK cua SALES_S5_OPPORTUNITY_PERMISSIONS.sql.txt - GO DUNG 17 dong role_permission cua 10 MA S5 qua yeu cau 2 nguoi (op=REMOVE) ROI xoa 10 ma.
-- VIET SAN, KHONG TU CHAY. Chay bang siscom_migrator: psql -X -v ON_ERROR_STOP=1 -v s5_sysadmin=<uuid> -v s5_approver=<uuid> -d <db> -f SALES_S5_OPPORTUNITY_PERMISSIONS_REMOVE.sql.txt
-- PHAM VI: CHI 10 ma liet ke tuong minh duoi day - KHONG dung LIKE 'opportunity.%'. Ma opportunity.* khac (them sau hoac them tay)
--   va moi quyen khong thuoc S5 duoc GIU NGUYEN; hau kiem so sanh dau van tay (md5) toan bo role_permission ngoai 10 ma truoc/sau.
-- TU CHOI (khong go gi) khi mot trong 10 ma dang duoc cap NGOAI ma tran S5 (vai tro khac, hoac scope khac) - co the la thay doi co duyet sau S5;
--   xu ly tay qua yeu cau 2 nguoi rieng roi chay lai. Liet ke dong ngoai ma tran trong thong diep.
-- Thu tu bat buoc: tung dong role_permission qua yeu cau REMOVE (guard trg_role_permission_guard chay ca khi xoa) -> DELETE permission 10 ma.
-- LUU Y: FK role_permission.permission_id la ON DELETE CASCADE (B1.2) => truoc DELETE permission PHAI tien kiem "khong con dong role_permission cua 10 ma" (buoc 2).
-- Sau file nay moi duoc chay SALES_S5_PERM_CLASS_ROLLBACK.sql.txt (file do tu choi khi con ma opportunity.*).
-- Dien tap CI: SALES_S5_OPPORTUNITY_PERMISSIONS_TEST.sql.txt (\ir voi s5_nested=1; ca tu choi chay voi ON_ERROR_STOP tat trong SAVEPOINT).
\if :{?s5_nested}
\else
\set ON_ERROR_STOP on
\endif
\pset pager off
\if :{?s5_sysadmin}
\else
\set s5_sysadmin ''
\endif
\if :{?s5_approver}
\else
\set s5_approver ''
\endif
SET client_encoding TO 'UTF8';
SELECT current_database() AS db_dich, current_user AS chay_bang;
\if :{?s5_nested}
\else
BEGIN;
\endif
-- 10 ma cua S5 (khop OPPORTUNITY_PERMISSIONS trong actor.ts - unit test opportunity-permissions.spec.ts kiem)
SELECT set_config('s5.sysadmin', :'s5_sysadmin', true), set_config('s5.approver', :'s5_approver', true),
       set_config('s5.codes', 'opportunity.read,opportunity.read_assigned,opportunity.create,opportunity.update,opportunity.move_stage,opportunity.close,opportunity.assign,opportunity.reopen,opportunity.delete,opportunity.restore', true);
DO $$
DECLARE v_sys uuid; v_app uuid;
BEGIN
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng (SET s1.allow_dev = yes trong cung phien).';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  IF NOT has_function_privilege(current_user, 'public.fn_auth_api_role_permission_change(uuid, uuid)', 'EXECUTE') THEN
    RAISE EXCEPTION 'Phai chay bang chu so huu (siscom_migrator).';
  END IF;
  IF public.fn_auth_bootstrap_open() THEN RAISE EXCEPTION 'Bootstrap con MO: khong co gi de go qua API dac quyen.'; END IF;
  BEGIN
    v_sys := nullif(current_setting('s5.sysadmin', true), '')::uuid; v_app := nullif(current_setting('s5.approver', true), '')::uuid;
  EXCEPTION WHEN OTHERS THEN RAISE EXCEPTION 's5_sysadmin / s5_approver phai la uuid.';
  END;
  IF v_sys IS NULL OR v_app IS NULL THEN RAISE EXCEPTION 'Thieu -v s5_sysadmin=<uuid> va/hoac -v s5_approver=<uuid>.'; END IF;
  IF v_sys = v_app THEN RAISE EXCEPTION 'TU CHOI: nguoi yeu cau va nguoi duyet phai KHAC nhau (2 nguoi).'; END IF;
END $$;

-- 0a. Quyen NGOAI ma tran S5 tren 10 ma => TU CHOI (khong go gi). Ma tran 17 dong = SALES_S5_OPPORTUNITY_PERMISSIONS.sql.txt (unit test giu 2 file khop).
SELECT (count(*) > 0) AS s5_rm_unexpected, coalesce(string_agg(x, ', ' ORDER BY x), '') AS s5_rm_unexpected_list FROM (
  SELECT ro.code || ':' || p.code || ':' || rp.scope::text AS x
  FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
  WHERE p.code = ANY (string_to_array(current_setting('s5.codes'), ','))
    AND (ro.code, p.code, rp.scope::text) NOT IN (SELECT m.role_code, m.perm_code, m.scope FROM (VALUES
      ('crm_admin',  'opportunity.read',          'ALL'),
      ('crm_admin',  'opportunity.create',        'ALL'),
      ('crm_admin',  'opportunity.update',        'ALL'),
      ('crm_admin',  'opportunity.move_stage',    'ALL'),
      ('crm_admin',  'opportunity.close',         'ALL'),
      ('crm_admin',  'opportunity.assign',        'ALL'),
      ('crm_admin',  'opportunity.reopen',        'ALL'),
      ('crm_admin',  'opportunity.delete',        'ALL'),
      ('crm_admin',  'opportunity.restore',       'ALL'),
      ('sales',      'opportunity.read',          'OWN'),
      ('sales',      'opportunity.create',        'OWN'),
      ('sales',      'opportunity.update',        'OWN'),
      ('sales',      'opportunity.move_stage',    'OWN'),
      ('sales',      'opportunity.close',         'OWN'),
      ('sales',      'opportunity.assign',        'OWN'),
      ('technician', 'opportunity.read_assigned', 'OWN'),
      ('accountant', 'opportunity.read',          'ALL')
    ) AS m(role_code, perm_code, scope))
) u \gset
\if :s5_rm_unexpected
\echo 'TU CHOI REMOVE: 10 ma S5 dang duoc cap NGOAI ma tran (co the la thay doi co duyet sau S5) - xu ly tay roi chay lai:' :s5_rm_unexpected_list
SELECT s5_remove_tu_choi_quyen_ngoai_ma_tran_s5();
\endif
-- 0b. Thong bao (khong chan): ma opportunity.* KHONG thuoc S5 se duoc giu nguyen
SELECT 'GIU NGUYEN (khong thuoc 10 ma S5): ' || string_agg(code, ', ' ORDER BY code) AS thong_bao
FROM public.permission WHERE code LIKE 'opportunity.%' AND code <> ALL (string_to_array(current_setting('s5.codes'), ',')) HAVING count(*) > 0;
-- 0c. Dau van tay moi role_permission NGOAI 10 ma (bao toan quyen khac)
SELECT set_config('s5.keep_fp', (SELECT coalesce(md5(string_agg(rp.role_id::text || '/' || rp.permission_id::text || '/' || rp.scope::text, ',' ORDER BY rp.role_id, rp.permission_id)), '-')
  FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id WHERE p.code <> ALL (string_to_array(current_setting('s5.codes'), ','))), true);

-- 1. Go TUNG dong role_permission cua 10 ma (moi dong mot yeu cau REMOVE)
DO $$
DECLARE v_sys uuid := current_setting('s5.sysadmin')::uuid; v_app uuid := current_setting('s5.approver')::uuid; r record; v_req uuid; n int := 0;
BEGIN
  FOR r IN SELECT rp.role_id, rp.permission_id, ro.code AS role_code, p.code AS perm_code
           FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
           WHERE p.code = ANY (string_to_array(current_setting('s5.codes'), ',')) ORDER BY ro.code, p.code
  LOOP
    v_req := public.fn_auth_api_request_create(v_sys, 'ROLE_PERMISSION_CHANGE', NULL, r.role_id,
               jsonb_build_object('op', 'REMOVE', 'permissionId', r.permission_id, 's5', r.role_code || ':' || r.perm_code), 60);
    PERFORM public.fn_auth_api_request_decide(v_app, v_req, true);
    PERFORM public.fn_auth_api_role_permission_change(v_sys, v_req);
    n := n + 1;
  END LOOP;
  RAISE NOTICE 'Sales S5 REMOVE: da go % dong role_permission cua 10 ma S5', n;
END $$;

-- 2. Tien kiem roi xoa 10 ma (FK la CASCADE => phai chac buoc 1 da sach, khong dua vao FK)
DO $$
DECLARE n int;
BEGIN
  SELECT count(*) INTO n FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id
   WHERE p.code = ANY (string_to_array(current_setting('s5.codes'), ','));
  IF n > 0 THEN RAISE EXCEPTION 'TU CHOI: con % dong role_permission cua 10 ma S5 sau buoc REMOVE - khong DELETE permission (FK CASCADE).', n; END IF;
END $$;
DELETE FROM public.permission WHERE code = ANY (string_to_array(current_setting('s5.codes'), ','));
-- 3. Hau kiem: 10 ma da het; quyen khac KHONG doi (dau van tay)
DO $$ BEGIN
  IF EXISTS (SELECT 1 FROM public.permission WHERE code = ANY (string_to_array(current_setting('s5.codes'), ','))) THEN
    RAISE EXCEPTION 'REMOVE chua sach: con ma S5 trong permission.';
  END IF;
  IF (SELECT coalesce(md5(string_agg(rp.role_id::text || '/' || rp.permission_id::text || '/' || rp.scope::text, ',' ORDER BY rp.role_id, rp.permission_id)), '-')
        FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id
       WHERE p.code <> ALL (string_to_array(current_setting('s5.codes'), ','))) <> current_setting('s5.keep_fp') THEN
    RAISE EXCEPTION 'REMOVE lam thay doi quyen NGOAI 10 ma S5 - huy toan bo.';
  END IF;
END $$;
\if :{?s5_nested}
\else
COMMIT;
\endif
SELECT 'SALES_S5_OPPORTUNITY_PERMISSIONS_REMOVE XONG' AS ket_qua;
