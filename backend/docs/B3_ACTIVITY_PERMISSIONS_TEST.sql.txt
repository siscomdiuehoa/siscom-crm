CREATE FUNCTION pg_temp.sod_try() RETURNS text LANGUAGE plpgsql AS $$
DECLARE v_req uuid;
BEGIN
  v_req := fn_auth_api_request_create('b3402000-0000-4000-8000-000000000001', 'ROLE_PERMISSION_CHANGE', NULL, 'b3401000-0000-4000-8000-000000000001',
             jsonb_build_object('op', 'ADD', 'permissionId', (SELECT id FROM permission WHERE code = 'activity.read'), 'scope', 'ALL'), 60);
  PERFORM fn_auth_api_request_decide('b3402000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('b3402000-0000-4000-8000-000000000001', v_req);
  RETURN 'CHAY DUOC';
EXCEPTION WHEN OTHERS THEN RETURN SQLSTATE;
END $$;
