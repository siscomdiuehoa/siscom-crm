-- Loi phai dung SQLSTATE VA thong bao bat dau bang tien to ma loi (ung dung doc tien to nay de tra dung ma HTTP).
CREATE FUNCTION pg_temp.expect_err(p_label text, p_stmt text, p_state text, p_prefix text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_stmt;
    INSERT INTO res VALUES (false, p_label || ' - KHONG bi chan');
  EXCEPTION WHEN OTHERS THEN
    IF SQLSTATE = p_state AND SQLERRM LIKE p_prefix || ':%' THEN INSERT INTO res VALUES (true, p_label);
    ELSE INSERT INTO res VALUES (false, p_label || ' - SQLSTATE ' || SQLSTATE || ' (' || SQLERRM || '), can ' || p_state || ' ' || p_prefix); END IF;
  END;
END $$;
CREATE FUNCTION pg_temp.expect_ok(p_label text, p_stmt text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN EXECUTE p_stmt; INSERT INTO res VALUES (true, p_label);
  EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')'); END;
END $$;
CREATE FUNCTION pg_temp.expect_true(p_label text, p_expr text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE r boolean;
BEGIN
  BEGIN EXECUTE 'SELECT (' || p_expr || ')' INTO r; INSERT INTO res VALUES (r IS TRUE, p_label);
  EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')'); END;
END $$;
