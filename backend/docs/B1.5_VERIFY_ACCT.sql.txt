-- SISCOM CRM - P1: KIEM quyen cua pool QUAN LY TAI KHOAN siscom_acct sau B1.5_ACCT_ROLE.sql.txt + B1.5_ACCT_GRANTS.sql.txt.
-- CHI DOC: moi cau thu nam trong transaction va ROLLBACK (WHERE false). In "PASS|n" / "FAIL|n". Ky vong hien hanh: PASS|26, khong FAIL.
-- CHAY BANG SUPERUSER (psql -tA -d <db thu> -f ...): siscom_acct KHONG co TEMP nen khong tu tao bang/ham tam; file tao bang/ham tam
--   bang superuser roi SET ROLE siscom_acct cho moi cau thu => quyen dung la cua siscom_acct (SET ROLE bo dac quyen superuser).
-- Guard: chi DB thu (regex) va khong phai baseline; phai co role siscom_acct.
-- Nhom L* = PHAI lam duoc (dung va du cho module accounts). Nhom M* = PHAI bi tu choi.
\set ON_ERROR_STOP on
\pset pager off
SET client_encoding TO 'UTF8';
SELECT CASE WHEN current_database() !~ '^siscom_crm_(restore|test)_[0-9]{8}_[0-9]{6}$' THEN 'SELECT s1_fail_khong_phai_db_thu()'
            WHEN current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN 'SELECT s1_fail_baseline_da_khoa()'
            WHEN NOT (SELECT rolsuper FROM pg_roles WHERE rolname = current_user) THEN 'SELECT s1_fail_phai_chay_bang_superuser()'
            WHEN NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'siscom_acct') THEN 'SELECT s1_fail_chua_co_role_siscom_acct()'
            WHEN to_regprocedure('public.fn_acct_api_account_audit(uuid, text, uuid, jsonb)') IS NULL THEN 'SELECT s1_fail_chua_ap_migration_20261124000000_p1_account_admin()'
            ELSE 'SELECT ''db thu: ' || current_database() || ' / ' || current_user || ' -> SET ROLE siscom_acct''' END \gexec

BEGIN;
CREATE TEMP TABLE s1_result (label text, ok boolean) ON COMMIT DROP;
GRANT INSERT ON s1_result TO siscom_acct;
CREATE FUNCTION pg_temp.expect_denied(p_label text, p_sql text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_sql;
    RAISE EXCEPTION 's1_undo' USING ERRCODE = 'P0001';
  EXCEPTION
    WHEN insufficient_privilege THEN INSERT INTO s1_result VALUES (p_label, true);
    WHEN raise_exception THEN INSERT INTO s1_result VALUES (p_label || CASE WHEN SQLERRM = 's1_undo' THEN ' [acct CHAY DUOC]' ELSE ' [' || SQLERRM || ']' END, false);
    WHEN OTHERS THEN INSERT INTO s1_result VALUES (p_label || ' [loi khac: ' || SQLSTATE || ']', false);
  END;
END $$;
CREATE FUNCTION pg_temp.expect_ok(p_label text, p_sql text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_sql;
    INSERT INTO s1_result VALUES (p_label, true);
  EXCEPTION WHEN OTHERS THEN INSERT INTO s1_result VALUES (p_label || ' [' || SQLSTATE || ']', false);
  END;
END $$;
-- Goi THAT mot ham qua quyen cua acct voi nguoi goi NULL: ham phai chay toi fn_auth_api_actor va tu choi bang tien to cua CHINH no
-- (APPROVAL_REQUIRED / BOOTSTRAP_OPEN). Thieu EXECUTE thi thong bao la "permission denied for function" => false.
CREATE FUNCTION pg_temp.probe_call() RETURNS boolean LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    PERFORM public.fn_acct_api_employee_create(NULL, 's1-probe', 's1-probe@example.invalid', NULL, NULL);
    RETURN false;
  EXCEPTION WHEN insufficient_privilege THEN RETURN SQLERRM LIKE 'APPROVAL_REQUIRED:%' OR SQLERRM LIKE 'BOOTSTRAP_OPEN:%';
            WHEN OTHERS THEN RETURN false;
  END;
END $$;
-- Tu day moi cau chay voi quyen cua siscom_acct (khong con superuser).
SET ROLE siscom_acct;
INSERT INTO s1_result SELECT 'L0 dang chay bang siscom_acct (SET ROLE), khong superuser',
  current_user = 'siscom_acct' AND NOT (SELECT rolsuper FROM pg_roles WHERE rolname = current_user);

-- L. PHAI LAM DUOC
SELECT pg_temp.expect_ok('L1 acct SELECT employee, employee_role, role, department',
  'SELECT (SELECT count(*) FROM public.employee) + (SELECT count(*) FROM public.employee_role) + (SELECT count(*) FROM public.role) + (SELECT count(*) FROM public.department)');
SELECT pg_temp.expect_ok('L2 acct SELECT 5 cot trang thai cua employee_credential',
  'SELECT employee_id, status, must_change_password, token_version, updated_at FROM public.employee_credential WHERE false');
INSERT INTO s1_result SELECT 'L3 acct EXECUTE du 12 ham (8 ham M1.2 + 4 ham P1)', count(*) = 12 AND bool_and(has_function_privilege('siscom_acct', f::regprocedure, 'EXECUTE'))
  FROM unnest(ARRAY[
    'public.fn_auth_api_request_create(uuid, public.privileged_change_type, uuid, uuid, jsonb, integer)', 'public.fn_auth_api_request_decide(uuid, uuid, boolean)',
    'public.fn_auth_api_role_grant(uuid, uuid)', 'public.fn_auth_api_role_revoke(uuid, uuid, uuid, uuid)',
    'public.fn_auth_api_reset_token_issue(uuid, uuid, text, integer)', 'public.fn_auth_api_reset_password(uuid, text, text, text, jsonb)',
    'public.fn_auth_api_credential_create(uuid, uuid)', 'public.fn_auth_api_credential_disable(uuid, uuid, text)',
    'public.fn_acct_api_employee_create(uuid, text, text, uuid, text)', 'public.fn_acct_api_employee_update(uuid, uuid, timestamptz, text, text, uuid, text)',
    'public.fn_acct_api_must_change_set(uuid, uuid, integer)', 'public.fn_acct_api_account_audit(uuid, text, uuid, jsonb)']) AS f;
INSERT INTO s1_result SELECT 'L4 acct co SET app.actor_employee_id / app.priv_request_id (ham SECURITY DEFINER can)',
  has_parameter_privilege('siscom_acct', 'app.actor_employee_id', 'SET') AND has_parameter_privilege('siscom_acct', 'app.priv_request_id', 'SET');
INSERT INTO s1_result SELECT 'L5 acct goi duoc fn_acct_api_employee_create (bi chinh ham tu choi vi nguoi goi NULL, khong phai thieu EXECUTE)', pg_temp.probe_call();

-- M. PHAI BI TU CHOI
SELECT pg_temp.expect_denied('M1 acct khong SELECT password_hash', 'SELECT password_hash FROM public.employee_credential WHERE false');
SELECT pg_temp.expect_denied('M2 acct khong SELECT password_params / password_algo', 'SELECT password_params, password_algo FROM public.employee_credential WHERE false');
SELECT pg_temp.expect_denied('M3 acct khong SELECT * employee_credential', 'SELECT * FROM public.employee_credential WHERE false');
INSERT INTO s1_result SELECT 'M4 acct khong SELECT bang nao khac (bang doc duoc = dung 5)',
  count(*) FILTER (WHERE has_any_column_privilege('siscom_acct', format('%I.%I', t.schemaname, t.tablename), 'SELECT')) = 5
  AND bool_and(NOT has_any_column_privilege('siscom_acct', format('%I.%I', t.schemaname, t.tablename), 'SELECT'))
        FILTER (WHERE t.tablename IN ('customer', 'contact', 'opportunity', 'activity', 'audit_log', 'auth_session', 'password_reset_token',
                                      'privileged_change_request', 'role_permission', 'permission', 'remember_device', 'app_meta', '_prisma_migrations'))
  FROM pg_tables t WHERE t.schemaname = 'public';
INSERT INTO s1_result SELECT 'M5 acct khong INSERT / UPDATE / DELETE / TRUNCATE bang nao',
  bool_and(NOT has_any_column_privilege('siscom_acct', format('%I.%I', t.schemaname, t.tablename), 'INSERT')
       AND NOT has_any_column_privilege('siscom_acct', format('%I.%I', t.schemaname, t.tablename), 'UPDATE')
       AND NOT has_table_privilege('siscom_acct', format('%I.%I', t.schemaname, t.tablename), 'DELETE')
       AND NOT has_table_privilege('siscom_acct', format('%I.%I', t.schemaname, t.tablename), 'TRUNCATE'))
  FROM pg_tables t WHERE t.schemaname = 'public';
SELECT pg_temp.expect_denied('M6 acct khong INSERT employee truc tiep (phai qua ham)', 'INSERT INTO public.employee (full_name) SELECT full_name FROM public.employee WHERE false');
SELECT pg_temp.expect_denied('M7 acct khong UPDATE employee truc tiep', 'UPDATE public.employee SET full_name = full_name WHERE false');
SELECT pg_temp.expect_denied('M8 acct khong INSERT employee_role (cap vai tro truc tiep)', 'INSERT INTO public.employee_role (employee_id, role_id) SELECT employee_id, role_id FROM public.employee_role WHERE false');
SELECT pg_temp.expect_denied('M9 acct khong UPDATE must_change_password truc tiep', 'UPDATE public.employee_credential SET must_change_password = must_change_password WHERE false');
SELECT pg_temp.expect_denied('M10 acct khong khoa dong credential (FOR UPDATE can quyen ghi)', 'SELECT employee_id FROM public.employee_credential WHERE false FOR UPDATE');
SELECT pg_temp.expect_denied('M11 acct khong fn_auth_api_change_own_password', 'SELECT public.fn_auth_api_change_own_password(NULL, 0, ''x'', ''x'', ''{}''::jsonb)');
SELECT pg_temp.expect_denied('M12 acct khong fn_auth_api_role_permission_change', 'SELECT public.fn_auth_api_role_permission_change(NULL, NULL)');
SELECT pg_temp.expect_denied('M13 acct khong fn_auth_bootstrap_close', 'SELECT public.fn_auth_bootstrap_close(''s1-probe'')');
SELECT pg_temp.expect_denied('M14 acct khong fn_auth_api_audit (ham noi bo)', 'SELECT public.fn_auth_api_audit(''USER'', NULL, ''X'', ''employee'', NULL, NULL, NULL)');
SELECT pg_temp.expect_denied('M15 acct khong fn_acct_api_employee_fields (ham noi bo P1)', 'SELECT * FROM public.fn_acct_api_employee_fields(''a'', ''a@b.c'', NULL, NULL, NULL)');
INSERT INTO s1_result SELECT 'M16 acct EXECUTE dung 12 ham SECURITY DEFINER, khong hon (' || count(*) || ' ham)', count(*) = 12
  FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace WHERE n.nspname = 'public' AND p.prosecdef AND has_function_privilege('siscom_acct', p.oid, 'EXECUTE');
SELECT pg_temp.expect_denied('M17 acct khong DDL', 'ALTER TABLE public.employee DISABLE TRIGGER ALL');
SELECT pg_temp.expect_denied('M18 acct khong CREATE TEMP TABLE', 'CREATE TEMP TABLE s1_probe_tmp (id int)');
INSERT INTO s1_result SELECT 'M19 acct khong dung sequence nao (khong dot ma nhan vien / khach hang)',
  NOT EXISTS (SELECT 1 FROM pg_sequences s WHERE s.schemaname = 'public'
    AND (has_sequence_privilege('siscom_acct', format('%I.%I', s.schemaname, s.sequencename), 'USAGE')
      OR has_sequence_privilege('siscom_acct', format('%I.%I', s.schemaname, s.sequencename), 'UPDATE')));
INSERT INTO s1_result SELECT 'M20 acct khong superuser/createrole/createdb/replication/bypassrls/inherit, khong chu so huu DB, khong CREATE public, khong TEMP, khong thanh vien role nao',
  (SELECT NOT rolsuper AND NOT rolcreaterole AND NOT rolcreatedb AND NOT rolreplication AND NOT rolbypassrls AND NOT rolinherit FROM pg_roles WHERE rolname = 'siscom_acct')
  AND (SELECT pg_get_userbyid(datdba) FROM pg_database WHERE datname = current_database()) <> 'siscom_acct'
  AND NOT has_schema_privilege('siscom_acct', 'public', 'CREATE') AND NOT has_database_privilege('siscom_acct', current_database(), 'TEMP')
  AND NOT EXISTS (SELECT 1 FROM pg_auth_members m WHERE m.member = (SELECT oid FROM pg_roles WHERE rolname = 'siscom_acct'));

RESET ROLE;
SELECT label || ' => ' || CASE WHEN ok THEN 'ok' ELSE 'KHONG DAT' END FROM s1_result ORDER BY label;
SELECT 'PASS|' || count(*) FROM s1_result WHERE ok IS TRUE;
SELECT 'FAIL|' || count(*) FROM s1_result WHERE ok IS NOT TRUE HAVING count(*) > 0;
ROLLBACK;
