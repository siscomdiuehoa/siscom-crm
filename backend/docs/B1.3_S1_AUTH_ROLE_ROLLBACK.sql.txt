-- SISCOM CRM - B1.4 P3c ROLLBACK ROLE XAC THUC: go quyen cua siscom_auth khoi DATABASE DICH, va chi go quyen CAP CLUSTER / DROP ROLE khi
-- role khong con duoc dung o DB nao khac. VIET SAN, chay bang SUPERUSER voi -v target_db=<ten>. Khong xoa du lieu, khong doi schema.
-- THU TU ROLLBACK P3c (xem B1.4_P3C_S3_AUTH_ROLE_DESIGN.md §7.2):
--   (1) trien khai lai backend BAN TRUOC P3c (ma P3c o production BAT BUOC AUTH_DATABASE_URL - khong the chi bo bien);
--   (2) chay B1.3_S1_APP_GRANTS_V2.sql.txt bang siscom_migrator (tra quyen xac thuc cho siscom_app; V2 KHONG dung toi siscom_auth);
--   (3) file nay (superuser) tren DB dich.
-- QA P3c I-1: KHONG dung DROP OWNED BY - no thu ca CONNECT cua MOI database va SET ON PARAMETER tren TOAN CLUSTER (nhu REASSIGN OWNED,
--   canh bao C1 cua ROLE_SPLIT) => DB that cung cluster mat pool auth. Thay bang REVOKE tuong minh trong DB dich; quyen cluster chi go khi
--   pg_shdepend khong con phu thuoc nao cua siscom_auth (buoc 2). CI dien tap voi DB canary cung cluster (pg-temp-db.sh unauth).
-- CACH CHAY: `psql -X -v ON_ERROR_STOP=1 -v target_db=<ten> -d <ten> -f <file nay>` (khong -X co the nap psqlrc). KHONG dung \i trong phien
--   tuong tac: voi ON_ERROR_ROLLBACK=on, loi DROP ROLE chi lui ve savepoint, transaction van mo voi REVOKE SET da chay (QA lan 4).
\set ON_ERROR_STOP on
\set ON_ERROR_ROLLBACK off
\pset pager off
\if :{?target_db}
\else
\set target_db ''
\endif
SET client_encoding TO 'UTF8';
SELECT current_database() AS db_dich, current_user AS chay_bang;
SELECT CASE WHEN :'target_db' = '' THEN 'SELECT s1_fail_thieu_target_db_can_psql_v_target_db_bang_ten_db_dich()'
            WHEN :'target_db' <> current_database() THEN 'SELECT s1_fail_target_db_khac_current_database()'
            ELSE 'SELECT ''target_db khop: ' || current_database() || '''' END \gexec
DO $$ BEGIN
  IF NOT (SELECT rolsuper FROM pg_roles WHERE rolname = current_user) THEN RAISE EXCEPTION 'Phai chay bang superuser.'; END IF;
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng.';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  -- Chay lai sau khi role da bi DROP => dung voi loi (fail-loud, co chu dich; QA N-3).
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'siscom_auth') THEN RAISE EXCEPTION 'Khong co role siscom_auth - khong co gi de go.'; END IF;
END $$;

-- 1. Trong DB dich: thu moi quyen tren doi tuong trong public (REVOKE ALL ON TABLE thu ca quyen theo cot), ham, schema, quyen mac dinh, CONNECT/TEMP.
BEGIN;
REVOKE ALL ON ALL TABLES IN SCHEMA public FROM siscom_auth;
REVOKE ALL ON ALL SEQUENCES IN SCHEMA public FROM siscom_auth;
DO $$
DECLARE f record;
BEGIN
  FOR f IN SELECT p.oid::regprocedure AS sig FROM pg_proc p JOIN pg_namespace ns ON ns.oid = p.pronamespace
           WHERE ns.nspname = 'public' AND NOT EXISTS (SELECT 1 FROM pg_depend d WHERE d.classid = 'pg_proc'::regclass AND d.objid = p.oid AND d.deptype = 'e') LOOP
    EXECUTE format('REVOKE ALL ON ROUTINE %s FROM siscom_auth', f.sig);
  END LOOP;
END $$;
REVOKE USAGE ON SCHEMA public FROM siscom_auth;
ALTER DEFAULT PRIVILEGES FOR ROLE siscom_migrator IN SCHEMA public REVOKE ALL ON TABLES FROM siscom_auth;
ALTER DEFAULT PRIVILEGES FOR ROLE siscom_migrator IN SCHEMA public REVOKE ALL ON SEQUENCES FROM siscom_auth;
SELECT format('REVOKE CONNECT, TEMP ON DATABASE %I FROM siscom_auth', current_database()) \gexec
COMMIT;
SELECT 'trong DB nay: quyen bang con lai = ' || count(*) FROM information_schema.table_privileges WHERE grantee = 'siscom_auth'
UNION ALL SELECT 'CONNECT DB nay = ' || has_database_privilege('siscom_auth', current_database(), 'CONNECT')::text;

-- 2. Cap cluster: CHI khi role khong con duoc dung o bat ky dau khac. Kiem bang pg_shdepend = MOI phu thuoc cua role tren toan cluster
--    (ACL muc database/bang/sequence/ham o DB khac, quyen so huu, ...). QA N-1 + QA lan 3 Q3-4 (dien tap PG tam, bang chung 10): chi xem
--    pg_database.datacl bo sot quyen MUC DOI TUONG o DB khac => DROP ROLE that bai SAU khi da REVOKE SET ON PARAMETER (nua chung). Vi vay:
--    (a) kiem pg_shdepend (tru pg_parameter_acl - thu ngay duoi), (b) REVOKE SET + DROP ROLE trong MOT transaction: that bai thi khong mat gi.
--    Neu "con dung" ma muon go het: chay file nay (buoc 1) tren tung DB duoc liet ke, lan cuoi se DROP ROLE.
CREATE TEMP TABLE s1_auth_other AS
  SELECT DISTINCT CASE WHEN s.dbid = (SELECT oid FROM pg_database WHERE datname = current_database())
                       THEN current_database() || ' (DB DICH: con sot ngoai buoc 1 - schema khac public / default privileges / so huu; go tay)'
                       ELSE coalesce(d.datname, d2.datname || ' (CONNECT/TEMP)', '(cluster: ' || s.classid::regclass::text || ')') END AS datname
  FROM pg_shdepend s LEFT JOIN pg_database d ON d.oid = s.dbid
       LEFT JOIN pg_database d2 ON s.classid = 'pg_database'::regclass AND d2.oid = s.objid
  WHERE s.refclassid = 'pg_authid'::regclass AND s.refobjid = (SELECT oid FROM pg_roles WHERE rolname = 'siscom_auth')
    AND s.classid <> 'pg_parameter_acl'::regclass;
SELECT 'DB/doi tuong khac con dung siscom_auth (pg_shdepend): ' || coalesce(string_agg(datname, ', '), '(khong)') FROM s1_auth_other;
BEGIN;
SELECT CASE WHEN EXISTS (SELECT 1 FROM s1_auth_other)
            THEN 'SELECT ''GIU role siscom_auth va quyen SET ON PARAMETER (con DB khac dung)'''
            ELSE 'REVOKE SET ON PARAMETER app.actor_employee_id, app.priv_request_id FROM siscom_auth' END \gexec
SELECT CASE WHEN EXISTS (SELECT 1 FROM s1_auth_other) THEN 'SELECT 1' ELSE 'DROP ROLE siscom_auth' END \gexec
COMMIT;
SELECT 'siscom_auth con ton tai: ' || EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'siscom_auth')::text;
