\set ON_ERROR_STOP on
\set ON_ERROR_ROLLBACK off
\pset pager off
SET client_encoding TO 'UTF8';
SELECT current_database() AS db_rollback, current_user AS chay_bang;
BEGIN;
DO $$ BEGIN
  IF current_database() !~ '^siscom_crm_(restore|test)_[0-9]{8}_[0-9]{6}$'
     OR current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % khong phai ban sao thu hop le cho rollback.', current_database();
  END IF;
  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'ck_customer_phone_e164' AND conrelid = 'public.customer'::regclass) THEN
    RAISE EXCEPTION 'Chua ap 20261104000000_customer_phone_optional (khong co ck_customer_phone_e164). Khong co gi de rollback.';
  END IF;

END $$;

ALTER TABLE public.customer DROP CONSTRAINT ck_customer_phone;
ALTER TABLE public.customer ADD CONSTRAINT ck_customer_phone CHECK (btrim(phone) <> '');
ALTER TABLE public.customer ALTER COLUMN phone SET NOT NULL;
COMMENT ON COLUMN public.customer.phone IS NULL;
DELETE FROM public._prisma_migrations WHERE migration_name = '20261104000000_customer_phone_optional';

DO $$ BEGIN
  IF EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'ck_customer_phone_e164' AND conrelid = 'public.customer'::regclass)
     OR EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema = 'public' AND table_name = 'customer' AND column_name = 'phone' AND is_nullable = 'YES')
     OR EXISTS (SELECT 1 FROM public._prisma_migrations WHERE migration_name = '20261104000000_customer_phone_optional') THEN

  END IF;
END $$;
COMMIT;
