-- VIET SAN, KHONG TU CHAY. Chay bang siscom_migrator: psql -X -v ON_ERROR_STOP=1 -v b3_sysadmin=<uuid> -v b3_approver=<uuid> -d <db> -f B3_ACTIVITY_PERMISSIONS_REMOVE.sql.txt
-- PHAM VI (ra soat doc lap 05/11/2026): CHI 9 ma liet ke tuong minh duoi day - KHONG dung LIKE 'activity.%'. Ma activity.* khac (A5+ hoac them tay)
--   va moi quyen khong thuoc A4 duoc GIU NGUYEN; hau kiem so sanh dau van tay (md5) toan bo role_permission ngoai 9 ma truoc/sau.
-- TU CHOI (khong go gi) khi mot trong 9 ma dang duoc cap NGOAI ma tran A4 (vai tro khac, hoac scope khac) - co the la thay doi co duyet sau A4;
--   xu ly tay qua yeu cau 2 nguoi rieng roi chay lai. Liet ke dong ngoai ma tran trong thong diep.
-- Thu tu bat buoc (thiet ke §5): tung dong role_permission qua yeu cau REMOVE (guard trg_role_permission_guard chay ca khi xoa) -> DELETE permission 9 ma.
-- LUU Y: FK role_permission.permission_id la ON DELETE CASCADE (B1.2), KHONG phai Restrict => truoc DELETE permission PHAI tien kiem "khong con dong
--   role_permission cua 9 ma" (buoc 2). Audit do cac ham API ghi.
-- Dien tap CI: B3_ACTIVITY_PERMISSIONS_TEST.sql.txt (\ir voi b3_nested=1; ca tu choi chay voi ON_ERROR_STOP tat trong SAVEPOINT).
\if :{?b3_nested}
\else
\set ON_ERROR_STOP on
\endif
\pset pager off
\if :{?b3_sysadmin}
\else
\set b3_sysadmin ''

\if :{?b3_approver}
\else

\endif
SET client_encoding TO 'UTF8';
SELECT current_database() AS db_dich, current_user AS chay_bang;
\if :{?b3_nested}
\else
BEGIN;
\endif
-- 9 ma cua A4 (khop ACTIVITY_PERMISSIONS trong actor.ts - unit test activity-permissions.spec.ts kiem)
SELECT set_config('b3.sysadmin', :'b3_sysadmin', true), set_config('b3.approver', :'b3_approver', true),
       set_config('b3.codes', 'activity.read,activity.create,activity.create_tech,activity.update,activity.update_own,activity.assign,activity.delete,activity.assign_external,activity.transfer_request', true);
DO $$
DECLARE v_sys uuid; v_app uuid;
BEGIN
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng (SET s1.allow_dev = yes trong cung phien).';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  IF NOT has_function_privilege(current_user, 'public.fn_auth_api_role_permission_change(uuid, uuid)', 'EXECUTE') THEN
    RAISE EXCEPTION 'Phai chay bang chu so huu (siscom_migrator).';
  END IF;
  IF public.fn_auth_bootstrap_open() THEN RAISE EXCEPTION 'Bootstrap con MO: khong co gi de go qua API dac quyen.'; END IF;
  BEGIN
    v_sys := nullif(current_setting('b3.sysadmin', true), '')::uuid; v_app := nullif(current_setting('b3.approver', true), '')::uuid;
  EXCEPTION WHEN OTHERS THEN RAISE EXCEPTION 'b3_sysadmin / b3_approver phai la uuid.';
  END;
  IF v_sys IS NULL OR v_app IS NULL THEN RAISE EXCEPTION 'Thieu -v b3_sysadmin=<uuid> va/hoac -v b3_approver=<uuid>.'; END IF;
  IF v_sys = v_app THEN RAISE EXCEPTION 'TU CHOI: nguoi yeu cau va nguoi duyet phai KHAC nhau (2 nguoi).'; END IF;
END $$;

-- 0a. Quyen NGOAI ma tran A4 tren 9 ma => TU CHOI (khong go gi). Ma tran 16 dong = B3_ACTIVITY_PERMISSIONS.sql.txt (unit test giu 2 file khop).
SELECT (count(*) > 0) AS b3_rm_unexpected, coalesce(string_agg(x, ', ' ORDER BY x), '') AS b3_rm_unexpected_list FROM (
  SELECT ro.code || ':' || p.code || ':' || rp.scope::text AS x
  FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
  WHERE p.code = ANY (string_to_array(current_setting('b3.codes'), ','))
    AND (ro.code, p.code, rp.scope::text) NOT IN (SELECT m.role_code, m.perm_code, m.scope FROM (VALUES
      ('crm_admin',  'activity.read',             'ALL'),
      ('crm_admin',  'activity.create',           'ALL'),
      ('crm_admin',  'activity.update',           'ALL'),
      ('crm_admin',  'activity.assign',           'ALL'),
      ('crm_admin',  'activity.delete',           'ALL'),
      ('crm_admin',  'activity.assign_external',  'ALL'),
      ('sales',      'activity.read',             'OWN'),
      ('sales',      'activity.create',           'OWN'),
      ('sales',      'activity.update',           'OWN'),
      ('sales',      'activity.assign',           'OWN'),
      ('sales',      'activity.assign_external',  'OWN'),
      ('technician', 'activity.read',             'DEPARTMENT'),
      ('technician', 'activity.create_tech',      'DEPARTMENT'),
      ('technician', 'activity.update_own',       'DEPARTMENT'),
      ('technician', 'activity.transfer_request', 'DEPARTMENT'),
      ('accountant', 'activity.read',             'ALL')
    ) AS m(role_code, perm_code, scope))
) u \gset
\if :b3_rm_unexpected
\echo 'TU CHOI REMOVE: 9 ma A4 dang duoc cap NGOAI ma tran (co the la thay doi co duyet sau A4) - xu ly tay roi chay lai:' :b3_rm_unexpected_list
SELECT b3_remove_tu_choi_quyen_ngoai_ma_tran_a4();
\endif
-- 0b. Thong bao (khong chan): ma activity.* KHONG thuoc A4 se duoc giu nguyen
SELECT 'GIU NGUYEN (khong thuoc 9 ma A4): ' || string_agg(code, ', ' ORDER BY code) AS thong_bao
FROM public.permission WHERE code LIKE 'activity.%' AND code <> ALL (string_to_array(current_setting('b3.codes'), ',')) HAVING count(*) > 0;
-- 0c. Dau van tay moi role_permission NGOAI 9 ma (bao toan quyen khac)
SELECT set_config('b3.keep_fp', (SELECT coalesce(md5(string_agg(rp.role_id::text || '/' || rp.permission_id::text || '/' || rp.scope::text, ',' ORDER BY rp.role_id, rp.permission_id)), '-')
  FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id WHERE p.code <> ALL (string_to_array(current_setting('b3.codes'), ','))), true);

-- 1. Go TUNG dong role_permission cua 9 ma (moi dong mot yeu cau REMOVE)
DO $$
DECLARE v_sys uuid := current_setting('b3.sysadmin')::uuid; v_app uuid := current_setting('b3.approver')::uuid; r record; v_req uuid; n int := 0;
BEGIN
  FOR r IN SELECT rp.role_id, rp.permission_id, ro.code AS role_code, p.code AS perm_code
           FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
           WHERE p.code = ANY (string_to_array(current_setting('b3.codes'), ',')) ORDER BY ro.code, p.code
  LOOP
    v_req := public.fn_auth_api_request_create(v_sys, 'ROLE_PERMISSION_CHANGE', NULL, r.role_id,
               jsonb_build_object('op', 'REMOVE', 'permissionId', r.permission_id, 'b3', r.role_code || ':' || r.perm_code), 60);
    PERFORM public.fn_auth_api_request_decide(v_app, v_req, true);
    PERFORM public.fn_auth_api_role_permission_change(v_sys, v_req);
    n := n + 1;
  END LOOP;
  RAISE NOTICE 'B3 A4 REMOVE: da go % dong role_permission cua 9 ma A4', n;
END $$;

-- 2. Tien kiem roi xoa 9 ma (FK la CASCADE => phai chac buoc 1 da sach, khong dua vao FK)
DO $$
DECLARE n int;
BEGIN
  SELECT count(*) INTO n FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id
   WHERE p.code = ANY (string_to_array(current_setting('b3.codes'), ','));
  IF n > 0 THEN RAISE EXCEPTION 'TU CHOI: con % dong role_permission cua 9 ma A4 sau buoc REMOVE - khong DELETE permission (FK CASCADE).', n; END IF;
END $$;
DELETE FROM public.permission WHERE code = ANY (string_to_array(current_setting('b3.codes'), ','));
-- 3. Hau kiem: 9 ma da het; quyen khac KHONG doi (dau van tay)
DO $$ BEGIN
  IF EXISTS (SELECT 1 FROM public.permission WHERE code = ANY (string_to_array(current_setting('b3.codes'), ','))) THEN
    RAISE EXCEPTION 'REMOVE chua sach: con ma A4 trong permission.';
  END IF;
  IF (SELECT coalesce(md5(string_agg(rp.role_id::text || '/' || rp.permission_id::text || '/' || rp.scope::text, ',' ORDER BY rp.role_id, rp.permission_id)), '-')
        FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id
       WHERE p.code <> ALL (string_to_array(current_setting('b3.codes'), ','))) <> current_setting('b3.keep_fp') THEN
    RAISE EXCEPTION 'REMOVE lam thay doi quyen NGOAI 9 ma A4 - huy toan bo.';
  END IF;
END $$;

COMMIT;
