### B.2 SQL chỉ đọc — đúng câu owner đã chạy (công cụ `KIEM-TRA-SQL-CHI-DOC.bat`)

\echo === Q2: chu so huu hien hanh (ky vong: 'E2E sale ...' -> NV-0003; 'E2E admin ...' -> trong) ===
SELECT c.code, left(c.name, 40) AS ten, e.code AS phu_trach
FROM customer c
LEFT JOIN customer_assignment a ON a.customer_id = c.id AND a.role = 'OWNER' AND a.ended_at IS NULL
LEFT JOIN employee e ON e.id = a.employee_id
WHERE c.deleted_at IS NULL ORDER BY c.code;
\echo === Q3: danh muc D3 (ky vong: customer_status 4, customer_tier 4, industry 10, source 5; khong co province) ===

\echo === Q6: thong tin dang nhap (ky vong: NV-0003 token_version > 1 va password_changed_at hom nay; failed_count 0; locked_until trong) ===
SELECT e.code, cr.token_version, cr.password_changed_at, cr.failed_count, cr.locked_until
FROM employee_credential cr JOIN employee e ON e.id = cr.employee_id ORDER BY e.code;
