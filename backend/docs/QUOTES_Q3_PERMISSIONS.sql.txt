-- 9 ma cua Q3 (khop QUOTE_PERMISSIONS trong actor.ts - unit test quote-permissions.spec.ts kiem)
SELECT set_config('q3.sysadmin', :'q3_sysadmin', true), set_config('q3.approver', :'q3_approver', true),
       set_config('q3.codes', 'quote.read,quote.create,quote.update,quote.send,quote.decide,quote.delete,quote.export,quote.template,quote.cost', true);
DO $$
DECLARE v_sys uuid; v_app uuid;
BEGIN
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng (SET s1.allow_dev = yes trong cung phien).';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  IF to_regprocedure('public.fn_auth_api_role_permission_change(uuid, uuid)') IS NULL THEN RAISE EXCEPTION 'Chua ap M1.2 (thieu fn_auth_api_role_permission_change).'; END IF;
  IF NOT has_function_privilege(current_user, 'public.fn_auth_api_role_permission_change(uuid, uuid)', 'EXECUTE') THEN
    RAISE EXCEPTION 'Phai chay bang chu so huu (siscom_migrator): % khong EXECUTE duoc ham API dac quyen.', current_user;
  END IF;
  IF public.fn_auth_perm_class('quote.read') <> 'BIZ' THEN
    RAISE EXCEPTION 'TU CHOI: quote.* chua thuoc nhom BIZ - ap migration 20261125000000_quotes truoc (SoD cho vai tro admin).';
  END IF;
  IF public.fn_auth_bootstrap_open() THEN
    RAISE EXCEPTION 'Bootstrap con MO: DB moi tinh lay ma tran tu bootstrap.ts (auth:bootstrap), khong dung file nay.';
  END IF;
  BEGIN
    v_sys := nullif(current_setting('q3.sysadmin', true), '')::uuid; v_app := nullif(current_setting('q3.approver', true), '')::uuid;
  EXCEPTION WHEN OTHERS THEN RAISE EXCEPTION 'q3_sysadmin / q3_approver phai la uuid (psql -v q3_sysadmin=... -v q3_approver=...).';
  END;
  IF v_sys IS NULL OR v_app IS NULL THEN RAISE EXCEPTION 'Thieu -v q3_sysadmin=<uuid> va/hoac -v q3_approver=<uuid>.'; END IF;
  IF v_sys = v_app THEN RAISE EXCEPTION 'TU CHOI: nguoi yeu cau va nguoi duyet phai KHAC nhau (2 nguoi).'; END IF;
  IF NOT EXISTS (SELECT 1 FROM public.role WHERE code IN ('crm_admin', 'sales', 'technician', 'accountant') HAVING count(*) = 4) THEN
    RAISE EXCEPTION 'Thieu vai tro crm_admin/sales/technician/accountant (DB chua bootstrap dung).';
  END IF;
END $$;

-- 1. Catalog quyen: 9 ma (bo qua neu da co)
INSERT INTO public.permission (code) SELECT unnest(string_to_array(current_setting('q3.codes'), ',')) ON CONFLICT (code) DO NOTHING;

-- 2. 18 dong role_permission theo ma tran muc 7.1 (docs/QUOTES_DESIGN.md) - moi dong mot yeu cau 2 nguoi; dong da co => bo qua
DO $$
DECLARE v_sys uuid := current_setting('q3.sysadmin')::uuid; v_app uuid := current_setting('q3.approver')::uuid;
        r record; v_role uuid; v_perm uuid; v_req uuid; n_added int := 0; n_skipped int := 0;
BEGIN
  FOR r IN SELECT * FROM (VALUES
      ('crm_admin',  'quote.read',     'ALL'),
      ('crm_admin',  'quote.create',   'ALL'),
      ('crm_admin',  'quote.update',   'ALL'),
      ('crm_admin',  'quote.send',     'ALL'),
      ('crm_admin',  'quote.decide',   'ALL'),
      ('crm_admin',  'quote.delete',   'ALL'),
      ('crm_admin',  'quote.export',   'ALL'),
      ('crm_admin',  'quote.template', 'ALL'),
      ('crm_admin',  'quote.cost',     'ALL'),
      ('sales',      'quote.read',     'OWN'),
      ('sales',      'quote.create',   'OWN'),
      ('sales',      'quote.update',   'OWN'),
      ('sales',      'quote.send',     'OWN'),
      ('sales',      'quote.decide',   'OWN'),
      ('sales',      'quote.delete',   'OWN'),
      ('sales',      'quote.export',   'OWN'),
      ('accountant', 'quote.read',     'ALL'),
      ('accountant', 'quote.export',   'ALL')
    ) AS m(role_code, perm_code, scope)
  LOOP
    SELECT id INTO v_role FROM public.role WHERE code = r.role_code;
    SELECT id INTO v_perm FROM public.permission WHERE code = r.perm_code;
    IF EXISTS (SELECT 1 FROM public.role_permission WHERE role_id = v_role AND permission_id = v_perm) THEN
      -- dong da co: bo qua CHI KHI scope dung ma tran; lech scope => tu choi (khong tu SET_SCOPE, xu ly tay co duyet)
      IF NOT EXISTS (SELECT 1 FROM public.role_permission WHERE role_id = v_role AND permission_id = v_perm AND scope::text = r.scope) THEN
        RAISE EXCEPTION 'TU CHOI: %:% da co nhung scope khac ma tran (%), khong tu doi.', r.role_code, r.perm_code, r.scope;
      END IF;
      n_skipped := n_skipped + 1; CONTINUE;
    END IF;
    v_req := public.fn_auth_api_request_create(v_sys, 'ROLE_PERMISSION_CHANGE', NULL, v_role,
               jsonb_build_object('op', 'ADD', 'permissionId', v_perm, 'scope', r.scope, 'q3', r.role_code || ':' || r.perm_code), 60);
    PERFORM public.fn_auth_api_request_decide(v_app, v_req, true);
    PERFORM public.fn_auth_api_role_permission_change(v_sys, v_req);
    n_added := n_added + 1;
  END LOOP;
  RAISE NOTICE 'Bao gia Q3: da cap % dong, bo qua % dong da co.', n_added, n_skipped;
END $$;

-- 3. Tom tat (ky vong: 9 ma; crm_admin 9, sales 7, accountant 2 = 18; technician 0; admin 0)
SELECT 'ma_quote (phai 9)' AS muc, count(*)::text AS gia_tri FROM public.permission WHERE code = ANY (string_to_array(current_setting('q3.codes'), ','))
UNION ALL SELECT 'dong_' || r.code || ' (phai ' || CASE r.code WHEN 'crm_admin' THEN 9 WHEN 'sales' THEN 7 WHEN 'accountant' THEN 2 ELSE 0 END || ')', count(p.id)::text
  FROM public.role r LEFT JOIN public.role_permission rp ON rp.role_id = r.id LEFT JOIN public.permission p ON p.id = rp.permission_id AND p.code = ANY (string_to_array(current_setting('q3.codes'), ','))
  WHERE r.code IN ('admin', 'crm_admin', 'sales', 'technician', 'accountant') GROUP BY r.code
UNION ALL SELECT 'tong_dong_quote (phai 18)', count(*)::text FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id WHERE p.code = ANY (string_to_array(current_setting('q3.codes'), ','));
