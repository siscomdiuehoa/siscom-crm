-- Dich: backend\docs\SALES_S5_OPPORTUNITY_PERMISSIONS.sql.txt (file moi, 17/11/2026). Thiet ke docs/SALES_DESIGN.md §5.2, §5.3. Mau: B3_ACTIVITY_PERMISSIONS.sql.txt.
-- Chay bang psql -f voi siscom_migrator (chu so huu; cac ham fn_auth_api_* da REVOKE FROM PUBLIC) tren DATABASE DICH:
--   psql -X -v ON_ERROR_STOP=1 -v s5_sysadmin=<uuid nhan vien nhom REQUEST (vai tro admin)> -v s5_approver=<uuid nhan vien co access.approve (crm_admin)> -d <db> -f SALES_S5_OPPORTUNITY_PERMISSIONS.sql.txt
-- Hai nguoi phai KHAC nhau, employee ACTIVE + credential ACTIVE (fn_auth_api_actor). Bootstrap phai DA DONG (DB moi tinh: khong dung file nay - ma tran BIZ trong bootstrap.ts).
-- TIEN DIEU KIEN: da ap migration 20261118000000_sales_perm_class (opportunity.* thuoc nhom BIZ => SoD cam vai tro admin giu cac ma nay). Chua ap => TU CHOI.
-- Moi dong role_permission = MOT yeu cau rieng: fn_auth_api_request_create (sysadmin, ROLE_PERMISSION_CHANGE, op=ADD) -> fn_auth_api_request_decide (approver)
--   -> fn_auth_api_role_permission_change (sysadmin). IDEMPOTENT: dong da ton tai DUNG scope thi BO QUA; lech scope => TU CHOI (khong tu SET_SCOPE); ma permission da co thi giu.
-- Audit: moi yeu cau/duyet/thuc thi ghi audit_log boi chinh cac ham API. admin (sysadmin) KHONG duoc cap (SoD). opportunity.export KHONG ton tai o dot nay (§5.2).
-- Rollback: SALES_S5_OPPORTUNITY_PERMISSIONS_REMOVE.sql.txt (op=REMOVE tung dong, roi xoa 10 ma). CHUA ap len DB staging/that nao (S12, Owner duyet rieng).
-- Dien tap CI: SALES_S5_OPPORTUNITY_PERMISSIONS_TEST.sql.txt (\ir file nay voi s5_nested=1: khong BEGIN/COMMIT rieng, nam trong transaction ngoai va ROLLBACK).

\pset pager off

\else

\else

SET client_encoding TO 'UTF8';
SELECT current_database() AS db_dich, current_user AS chay_bang;
\if :{?s5_nested}

BEGIN;
\endif
-- 10 ma cua S5 (khop OPPORTUNITY_PERMISSIONS trong actor.ts - unit test opportunity-permissions.spec.ts kiem)
SELECT set_config('s5.sysadmin', :'s5_sysadmin', true), set_config('s5.approver', :'s5_approver', true),
       set_config('s5.codes', 'opportunity.read,opportunity.read_assigned,opportunity.create,opportunity.update,opportunity.move_stage,opportunity.close,opportunity.assign,opportunity.reopen,opportunity.delete,opportunity.restore', true);
DO $$
DECLARE v_sys uuid; v_app uuid;
BEGIN
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng (SET s1.allow_dev = yes trong cung phien).';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  IF to_regprocedure('public.fn_auth_api_role_permission_change(uuid, uuid)') IS NULL THEN RAISE EXCEPTION 'Chua ap M1.2 (thieu fn_auth_api_role_permission_change).'; END IF;
  IF NOT has_function_privilege(current_user, 'public.fn_auth_api_role_permission_change(uuid, uuid)', 'EXECUTE') THEN
    RAISE EXCEPTION 'Phai chay bang chu so huu (siscom_migrator): % khong EXECUTE duoc ham API dac quyen.', current_user;
  END IF;
  IF public.fn_auth_perm_class('opportunity.read') <> 'BIZ' THEN
    RAISE EXCEPTION 'TU CHOI: opportunity.* chua thuoc nhom BIZ - ap migration 20261118000000_sales_perm_class truoc (SoD cho vai tro admin).';
  END IF;
  IF public.fn_auth_bootstrap_open() THEN
    RAISE EXCEPTION 'Bootstrap con MO: DB moi tinh lay ma tran tu bootstrap.ts (auth:bootstrap), khong dung file nay.';
  END IF;
  BEGIN
    v_sys := nullif(current_setting('s5.sysadmin', true), '')::uuid; v_app := nullif(current_setting('s5.approver', true), '')::uuid;
  EXCEPTION WHEN OTHERS THEN RAISE EXCEPTION 's5_sysadmin / s5_approver phai la uuid (psql -v s5_sysadmin=... -v s5_approver=...).';
  END;
  IF v_sys IS NULL OR v_app IS NULL THEN RAISE EXCEPTION 'Thieu -v s5_sysadmin=<uuid> va/hoac -v s5_approver=<uuid>.'; END IF;
  IF v_sys = v_app THEN RAISE EXCEPTION 'TU CHOI: nguoi yeu cau va nguoi duyet phai KHAC nhau (2 nguoi).'; END IF;
  IF NOT EXISTS (SELECT 1 FROM public.role WHERE code IN ('crm_admin', 'sales', 'technician', 'accountant') HAVING count(*) = 4) THEN
    RAISE EXCEPTION 'Thieu vai tro crm_admin/sales/technician/accountant (DB chua bootstrap dung).';
  END IF;
END $$;

-- 1. Catalog quyen: 10 ma (bo qua neu da co)
INSERT INTO public.permission (code) SELECT unnest(string_to_array(current_setting('s5.codes'), ',')) ON CONFLICT (code) DO NOTHING;

-- 2. 17 dong role_permission theo ma tran §5.2 - moi dong mot yeu cau 2 nguoi; dong da co => bo qua
DO $$
DECLARE v_sys uuid := current_setting('s5.sysadmin')::uuid; v_app uuid := current_setting('s5.approver')::uuid;
        r record; v_role uuid; v_perm uuid; v_req uuid; n_added int := 0; n_skipped int := 0;
BEGIN
  FOR r IN SELECT * FROM (VALUES
      ('crm_admin',  'opportunity.read',          'ALL'),
      ('crm_admin',  'opportunity.create',        'ALL'),
      ('crm_admin',  'opportunity.update',        'ALL'),
      ('crm_admin',  'opportunity.move_stage',    'ALL'),
      ('crm_admin',  'opportunity.close',         'ALL'),
      ('crm_admin',  'opportunity.assign',        'ALL'),
      ('crm_admin',  'opportunity.reopen',        'ALL'),
      ('crm_admin',  'opportunity.delete',        'ALL'),
      ('crm_admin',  'opportunity.restore',       'ALL'),
      ('sales',      'opportunity.read',          'OWN'),
      ('sales',      'opportunity.create',        'OWN'),
      ('sales',      'opportunity.update',        'OWN'),
      ('sales',      'opportunity.move_stage',    'OWN'),
      ('sales',      'opportunity.close',         'OWN'),
      ('sales',      'opportunity.assign',        'OWN'),
      ('technician', 'opportunity.read_assigned', 'OWN'),
      ('accountant', 'opportunity.read',          'ALL')
    ) AS m(role_code, perm_code, scope)
  LOOP
    SELECT id INTO v_role FROM public.role WHERE code = r.role_code;
    SELECT id INTO v_perm FROM public.permission WHERE code = r.perm_code;
    IF EXISTS (SELECT 1 FROM public.role_permission WHERE role_id = v_role AND permission_id = v_perm) THEN
      -- dong da co: bo qua CHI KHI scope dung ma tran; lech scope => tu choi (khong tu SET_SCOPE, xu ly tay co duyet)
      IF NOT EXISTS (SELECT 1 FROM public.role_permission WHERE role_id = v_role AND permission_id = v_perm AND scope::text = r.scope) THEN
        RAISE EXCEPTION 'TU CHOI: %:% da co nhung scope khac ma tran (%), khong tu doi.', r.role_code, r.perm_code, r.scope;
      END IF;
      n_skipped := n_skipped + 1; CONTINUE;
    END IF;
    v_req := public.fn_auth_api_request_create(v_sys, 'ROLE_PERMISSION_CHANGE', NULL, v_role,
               jsonb_build_object('op', 'ADD', 'permissionId', v_perm, 'scope', r.scope, 's5', r.role_code || ':' || r.perm_code), 60);
    PERFORM public.fn_auth_api_request_decide(v_app, v_req, true);
    PERFORM public.fn_auth_api_role_permission_change(v_sys, v_req);
    n_added := n_added + 1;
  END LOOP;
  RAISE NOTICE 'Sales S5: da cap % dong, bo qua % dong da co.', n_added, n_skipped;
END $$;

-- 3. Tom tat (ky vong: 10 ma; crm_admin 9, sales 6, technician 1, accountant 1 = 17; admin 0)
SELECT 'ma_opportunity (phai 10)' AS muc, count(*)::text AS gia_tri FROM public.permission WHERE code = ANY (string_to_array(current_setting('s5.codes'), ','))
UNION ALL SELECT 'dong_' || r.code || ' (phai ' || CASE r.code WHEN 'crm_admin' THEN 9 WHEN 'sales' THEN 6 WHEN 'technician' THEN 1 WHEN 'accountant' THEN 1 ELSE 0 END || ')', count(p.id)::text
  FROM public.role r LEFT JOIN public.role_permission rp ON rp.role_id = r.id LEFT JOIN public.permission p ON p.id = rp.permission_id AND p.code = ANY (string_to_array(current_setting('s5.codes'), ','))
  WHERE r.code IN ('admin', 'crm_admin', 'sales', 'technician', 'accountant') GROUP BY r.code
UNION ALL SELECT 'tong_dong_opportunity (phai 17)', count(*)::text FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id WHERE p.code = ANY (string_to_array(current_setting('s5.codes'), ','));
\if :{?s5_nested}
\else
COMMIT;
\endif
SELECT 'SALES_S5_OPPORTUNITY_PERMISSIONS XONG' AS ket_qua;
