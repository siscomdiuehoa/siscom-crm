-- 9 ma cua Q3 (khop QUOTE_PERMISSIONS trong actor.ts - unit test quote-permissions.spec.ts kiem)
SELECT set_config('q3.sysadmin', :'q3_sysadmin', true), set_config('q3.approver', :'q3_approver', true),
       set_config('q3.codes', 'quote.read,quote.create,quote.update,quote.send,quote.decide,quote.delete,quote.export,quote.template,quote.cost', true);
DO $$
DECLARE v_sys uuid; v_app uuid;
BEGIN
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng (SET s1.allow_dev = yes trong cung phien).';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  IF NOT has_function_privilege(current_user, 'public.fn_auth_api_role_permission_change(uuid, uuid)', 'EXECUTE') THEN
    RAISE EXCEPTION 'Phai chay bang chu so huu (siscom_migrator).';
  END IF;
  IF public.fn_auth_bootstrap_open() THEN RAISE EXCEPTION 'Bootstrap con MO: khong co gi de go qua API dac quyen.'; END IF;
  BEGIN
    v_sys := nullif(current_setting('q3.sysadmin', true), '')::uuid; v_app := nullif(current_setting('q3.approver', true), '')::uuid;
  EXCEPTION WHEN OTHERS THEN RAISE EXCEPTION 'q3_sysadmin / q3_approver phai la uuid.';
  END;
  IF v_sys IS NULL OR v_app IS NULL THEN RAISE EXCEPTION 'Thieu -v q3_sysadmin=<uuid> va/hoac -v q3_approver=<uuid>.'; END IF;
  IF v_sys = v_app THEN RAISE EXCEPTION 'TU CHOI: nguoi yeu cau va nguoi duyet phai KHAC nhau (2 nguoi).'; END IF;
END $$;

-- Ma tran 18 dong = QUOTES_Q3_PERMISSIONS.sql.txt (unit test giu 2 file khop). MOT ban duy nhat trong file nay (bang tam), dung cho

-- 0a. Quyen NGOAI ma tran Q3 tren 9 ma (vai tro khac HOAC scope khac) => TU CHOI (khong go gi)
SELECT (count(*) > 0) AS q3_rm_unexpected, coalesce(string_agg(x, ', ' ORDER BY x), '') AS q3_rm_unexpected_list FROM (
  SELECT ro.code || ':' || p.code || ':' || rp.scope::text AS x
  FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
  WHERE p.code = ANY (string_to_array(current_setting('q3.codes'), ','))
    AND NOT EXISTS (SELECT 1 FROM q3_matrix m WHERE m.role_code = ro.code AND m.perm_code = p.code AND m.scope = rp.scope::text)
) u \gset
\if :q3_rm_unexpected
\echo 'TU CHOI REMOVE: 9 ma Q3 dang duoc cap NGOAI ma tran (co the la thay doi co duyet sau Q3) - xu ly tay roi chay lai:' :q3_rm_unexpected_list
SELECT q3_remove_tu_choi_quyen_ngoai_ma_tran_q3();
\endif
-- 0b. Thong bao (khong chan): ma quote.* KHONG thuoc Q3 se duoc giu nguyen
SELECT 'GIU NGUYEN (khong thuoc 9 ma Q3): ' || string_agg(code, ', ' ORDER BY code) AS thong_bao
FROM public.permission WHERE code LIKE 'quote.%' AND code <> ALL (string_to_array(current_setting('q3.codes'), ',')) HAVING count(*) > 0;
-- 0c. Dau van tay moi role_permission NGOAI 9 ma (bao toan quyen khac)
SELECT set_config('q3.keep_fp', (SELECT coalesce(md5(string_agg(rp.role_id::text || '/' || rp.permission_id::text || '/' || rp.scope::text, ',' ORDER BY rp.role_id, rp.permission_id)), '-')
  FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id WHERE p.code <> ALL (string_to_array(current_setting('q3.codes'), ','))), true);

-- 1. Go TUNG dong role_permission cua 9 ma (moi dong mot yeu cau REMOVE). Kiem LAI "khong co dong ngoai ma tran" ngay trong khoi nay va
--    vong go CHI lay dong KHOP ma tran (vai tro + ma + scope): du co loi khoa, dong ngoai ma tran khong bao gio bi go o day.
DO $$
DECLARE v_sys uuid := current_setting('q3.sysadmin')::uuid; v_app uuid := current_setting('q3.approver')::uuid; r record; v_req uuid; n int := 0; v_bad text;
BEGIN
  SELECT string_agg(ro.code || ':' || p.code || ':' || rp.scope::text, ', ' ORDER BY ro.code, p.code) INTO v_bad
    FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
   WHERE p.code = ANY (string_to_array(current_setting('q3.codes'), ','))
     AND NOT EXISTS (SELECT 1 FROM q3_matrix m WHERE m.role_code = ro.code AND m.perm_code = p.code AND m.scope = rp.scope::text);
  IF v_bad IS NOT NULL THEN RAISE EXCEPTION 'TU CHOI REMOVE: co dong ngoai ma tran Q3 tai thoi diem go: % - khong go gi.', v_bad; END IF;
  FOR r IN SELECT rp.role_id, rp.permission_id, ro.code AS role_code, p.code AS perm_code
           FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id JOIN public.role ro ON ro.id = rp.role_id
           JOIN q3_matrix m ON m.role_code = ro.code AND m.perm_code = p.code AND m.scope = rp.scope::text
           WHERE p.code = ANY (string_to_array(current_setting('q3.codes'), ',')) ORDER BY ro.code, p.code
  LOOP
    v_req := public.fn_auth_api_request_create(v_sys, 'ROLE_PERMISSION_CHANGE', NULL, r.role_id,
               jsonb_build_object('op', 'REMOVE', 'permissionId', r.permission_id, 'q3', r.role_code || ':' || r.perm_code), 60);
    PERFORM public.fn_auth_api_request_decide(v_app, v_req, true);
    PERFORM public.fn_auth_api_role_permission_change(v_sys, v_req);
    n := n + 1;
  END LOOP;
  RAISE NOTICE 'Bao gia Q3 REMOVE: da go % dong role_permission cua 9 ma Q3', n;
END $$;

-- 2. Tien kiem roi xoa 9 ma (FK la CASCADE => phai chac buoc 1 da sach, khong dua vao FK)
DO $$
DECLARE n int;
BEGIN
  SELECT count(*) INTO n FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id
   WHERE p.code = ANY (string_to_array(current_setting('q3.codes'), ','));
  IF n > 0 THEN RAISE EXCEPTION 'TU CHOI: con % dong role_permission cua 9 ma Q3 sau buoc REMOVE - khong DELETE permission (FK CASCADE).', n; END IF;
END $$;
DELETE FROM public.permission WHERE code = ANY (string_to_array(current_setting('q3.codes'), ','));
-- 3. Hau kiem: 9 ma da het; quyen khac KHONG doi (dau van tay)
DO $$ BEGIN
  IF EXISTS (SELECT 1 FROM public.permission WHERE code = ANY (string_to_array(current_setting('q3.codes'), ','))) THEN
    RAISE EXCEPTION 'REMOVE chua sach: con ma Q3 trong permission.';
  END IF;
  IF (SELECT coalesce(md5(string_agg(rp.role_id::text || '/' || rp.permission_id::text || '/' || rp.scope::text, ',' ORDER BY rp.role_id, rp.permission_id)), '-')
        FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id
       WHERE p.code <> ALL (string_to_array(current_setting('q3.codes'), ','))) <> current_setting('q3.keep_fp') THEN
    RAISE EXCEPTION 'REMOVE lam thay doi quyen NGOAI 9 ma Q3 - huy toan bo.';
  END IF;
END $$;
