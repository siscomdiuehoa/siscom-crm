-- Dich: D:\SISCOM-CRM\backend\docs\B3_ACTIVITY_PERMISSIONS.sql.txt (file moi, 05/11/2026). Thiet ke docs/B3_ACTIVITIES_DESIGN.md §5 ("Cap quyen cho DB da bootstrap").
-- Chay bang psql -f voi siscom_migrator (chu so huu; cac ham fn_auth_api_* da REVOKE FROM PUBLIC) tren DATABASE DICH:
--   psql -X -v ON_ERROR_STOP=1 -v b3_sysadmin=<uuid nhan vien nhom REQUEST (vai tro admin)> -v b3_approver=<uuid nhan vien co access.approve (crm_admin)> -d <db> -f B3_ACTIVITY_PERMISSIONS.sql.txt
-- Hai nguoi phai KHAC nhau, employee ACTIVE + credential ACTIVE (fn_auth_api_actor). Bootstrap phai DA DONG (DB moi tinh: khong dung file nay - ma tran BIZ trong bootstrap.ts).
-- Moi dong role_permission = MOT yeu cau rieng: fn_auth_api_request_create (sysadmin, ROLE_PERMISSION_CHANGE, op=ADD) -> fn_auth_api_request_decide (approver)
--   -> fn_auth_api_role_permission_change (sysadmin). IDEMPOTENT: dong da ton tai DUNG scope thi BO QUA (khong goi lai ham); lech scope => TU CHOI (khong tu SET_SCOPE); ma permission da co thi giu.
-- Audit: moi yeu cau/duyet/thuc thi ghi audit_log boi chinh cac ham API (PRIV_REQUEST_*, ROLE_PERMISSION_CHANGE). admin (sysadmin) KHONG duoc cap BIZ (SoD).
-- Rollback: B3_ACTIVITY_PERMISSIONS_REMOVE.sql.txt (op=REMOVE tung dong, roi xoa 9 ma). CHUA ap len DB demo/that nao (A9, anh duyet rieng).
-- Dien tap CI: B3_ACTIVITY_PERMISSIONS_TEST.sql.txt (\ir file nay voi b3_nested=1: khong BEGIN/COMMIT rieng, nam trong transaction ngoai va ROLLBACK).
\set ON_ERROR_STOP on
\pset pager off
\if :{?b3_sysadmin}
\else
\set b3_sysadmin ''
\endif
\if :{?b3_approver}
\else
\set b3_approver ''
\endif
SET client_encoding TO 'UTF8';
SELECT current_database() AS db_dich, current_user AS chay_bang;
\if :{?b3_nested}
\else
BEGIN;
\endif
SELECT set_config('b3.sysadmin', :'b3_sysadmin', true), set_config('b3.approver', :'b3_approver', true);
DO $$
DECLARE v_sys uuid; v_app uuid;
BEGIN
  IF current_database() = 'siscom_crm_dev' AND coalesce(current_setting('s1.allow_dev', true), '') <> 'yes' THEN
    RAISE EXCEPTION 'TU CHOI: siscom_crm_dev. Can phe duyet rieng (SET s1.allow_dev = yes trong cung phien).';
  END IF;
  IF current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % la baseline da khoa.', current_database();
  END IF;
  IF to_regprocedure('public.fn_auth_api_role_permission_change(uuid, uuid)') IS NULL THEN RAISE EXCEPTION 'Chua ap M1.2 (thieu fn_auth_api_role_permission_change).'; END IF;
  IF NOT has_function_privilege(current_user, 'public.fn_auth_api_role_permission_change(uuid, uuid)', 'EXECUTE') THEN
    RAISE EXCEPTION 'Phai chay bang chu so huu (siscom_migrator): % khong EXECUTE duoc ham API dac quyen.', current_user;
  END IF;
  IF public.fn_auth_bootstrap_open() THEN
    RAISE EXCEPTION 'Bootstrap con MO: DB moi tinh lay ma tran tu bootstrap.ts (auth:bootstrap), khong dung file nay.';
  END IF;
  BEGIN
    v_sys := nullif(current_setting('b3.sysadmin', true), '')::uuid; v_app := nullif(current_setting('b3.approver', true), '')::uuid;
  EXCEPTION WHEN OTHERS THEN RAISE EXCEPTION 'b3_sysadmin / b3_approver phai la uuid (psql -v b3_sysadmin=... -v b3_approver=...).';
  END;
  IF v_sys IS NULL OR v_app IS NULL THEN RAISE EXCEPTION 'Thieu -v b3_sysadmin=<uuid> va/hoac -v b3_approver=<uuid>.'; END IF;
  IF v_sys = v_app THEN RAISE EXCEPTION 'TU CHOI: nguoi yeu cau va nguoi duyet phai KHAC nhau (2 nguoi).'; END IF;
  IF NOT EXISTS (SELECT 1 FROM public.role WHERE code IN ('crm_admin', 'sales', 'technician', 'accountant') HAVING count(*) = 4) THEN
    RAISE EXCEPTION 'Thieu vai tro crm_admin/sales/technician/accountant (DB chua bootstrap dung).';
  END IF;
END $$;

-- 1. Catalog quyen: 9 ma (khong co trigger chan INSERT permission; bo qua neu da co)
INSERT INTO public.permission (code) VALUES
 ('activity.read'), ('activity.create'), ('activity.create_tech'), ('activity.update'), ('activity.update_own'),
 ('activity.assign'), ('activity.delete'), ('activity.assign_external'), ('activity.transfer_request')
ON CONFLICT (code) DO NOTHING;

-- 2. 16 dong role_permission theo ma tran §5 (B3 v5.3) - moi dong mot yeu cau 2 nguoi; dong da co => bo qua
DO $$
DECLARE v_sys uuid := current_setting('b3.sysadmin')::uuid; v_app uuid := current_setting('b3.approver')::uuid;
        r record; v_role uuid; v_perm uuid; v_req uuid; n_added int := 0; n_skipped int := 0;
BEGIN
  FOR r IN SELECT * FROM (VALUES
      ('crm_admin',  'activity.read',             'ALL'),
      ('crm_admin',  'activity.create',           'ALL'),
      ('crm_admin',  'activity.update',           'ALL'),
      ('crm_admin',  'activity.assign',           'ALL'),
      ('crm_admin',  'activity.delete',           'ALL'),
      ('crm_admin',  'activity.assign_external',  'ALL'),
      ('sales',      'activity.read',             'OWN'),
      ('sales',      'activity.create',           'OWN'),
      ('sales',      'activity.update',           'OWN'),
      ('sales',      'activity.assign',           'OWN'),
      ('sales',      'activity.assign_external',  'OWN'),
      ('technician', 'activity.read',             'DEPARTMENT'),
      ('technician', 'activity.create_tech',      'DEPARTMENT'),
      ('technician', 'activity.update_own',       'DEPARTMENT'),
      ('technician', 'activity.transfer_request', 'DEPARTMENT'),
      ('accountant', 'activity.read',             'ALL')
    ) AS m(role_code, perm_code, scope)
  LOOP
    SELECT id INTO v_role FROM public.role WHERE code = r.role_code;
    SELECT id INTO v_perm FROM public.permission WHERE code = r.perm_code;
    IF EXISTS (SELECT 1 FROM public.role_permission WHERE role_id = v_role AND permission_id = v_perm) THEN
      -- dong da co: bo qua CHI KHI scope dung ma tran; lech scope => tu choi (khong tu SET_SCOPE, xu ly tay co duyet)
      IF NOT EXISTS (SELECT 1 FROM public.role_permission WHERE role_id = v_role AND permission_id = v_perm AND scope::text = r.scope) THEN
        RAISE EXCEPTION 'TU CHOI: %:% da co nhung scope khac ma tran (%), khong tu doi.', r.role_code, r.perm_code, r.scope;
      END IF;
      n_skipped := n_skipped + 1; CONTINUE;
    END IF;
    v_req := public.fn_auth_api_request_create(v_sys, 'ROLE_PERMISSION_CHANGE', NULL, v_role,
               jsonb_build_object('op', 'ADD', 'permissionId', v_perm, 'scope', r.scope, 'b3', r.role_code || ':' || r.perm_code), 60);
    PERFORM public.fn_auth_api_request_decide(v_app, v_req, true);
    PERFORM public.fn_auth_api_role_permission_change(v_sys, v_req);
    n_added := n_added + 1;
  END LOOP;
  RAISE NOTICE 'B3 A4: da cap % dong, bo qua % dong da co.', n_added, n_skipped;
END $$;

-- 3. Tom tat (ky vong: 9 ma; crm_admin 6, sales 5, technician 4, accountant 1 = 16; admin 0)
SELECT 'ma_activity (phai 9)' AS muc, count(*)::text AS gia_tri FROM public.permission WHERE code LIKE 'activity.%'
UNION ALL SELECT 'dong_' || r.code || ' (phai ' || CASE r.code WHEN 'crm_admin' THEN 6 WHEN 'sales' THEN 5 WHEN 'technician' THEN 4 WHEN 'accountant' THEN 1 ELSE 0 END || ')', count(p.id)::text
  FROM public.role r LEFT JOIN public.role_permission rp ON rp.role_id = r.id LEFT JOIN public.permission p ON p.id = rp.permission_id AND p.code LIKE 'activity.%'
  WHERE r.code IN ('admin', 'crm_admin', 'sales', 'technician', 'accountant') GROUP BY r.code
UNION ALL SELECT 'tong_dong_activity (phai 16)', count(*)::text FROM public.role_permission rp JOIN public.permission p ON p.id = rp.permission_id WHERE p.code LIKE 'activity.%';
\if :{?b3_nested}
\else
COMMIT;
\endif
SELECT 'B3_ACTIVITY_PERMISSIONS XONG' AS ket_qua;
