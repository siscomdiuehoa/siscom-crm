DO $$ BEGIN
  IF current_database() !~ '^siscom_crm_(restore|test)_[0-9]{8}_[0-9]{6}$'
     OR current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % khong phai ban sao thu hop le.', current_database();

  END IF;
  IF EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'ck_contact_reach' AND conrelid = 'public.contact'::regclass) THEN
    RAISE EXCEPTION 'Chua ap migration 20261104000100_contact_reach_optional.';
  END IF;
END $$;
BEGIN;
CREATE TEMP TABLE res (ok boolean, label text) ON COMMIT DROP;
CREATE FUNCTION pg_temp.expect_err(p_label text, p_sql text, p_state text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_sql;
    RAISE EXCEPTION 'pr203_undo' USING ERRCODE = 'P0001';
  EXCEPTION
    WHEN raise_exception THEN INSERT INTO res VALUES (false, p_label || CASE WHEN SQLERRM = 'pr203_undo' THEN ' [CHAY DUOC]' ELSE ' [' || SQLERRM || ']' END);
    WHEN OTHERS THEN INSERT INTO res VALUES (SQLSTATE = p_state, p_label || CASE WHEN SQLSTATE = p_state THEN '' ELSE ' [' || SQLSTATE || ' thay vi ' || p_state || ']' END);
  END;
END $$;
CREATE FUNCTION pg_temp.expect_ok(p_label text, p_sql text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  EXECUTE p_sql;
  INSERT INTO res VALUES (true, p_label);
EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' [' || SQLSTATE || ' ' || SQLERRM || ']');
END $$;
CREATE FUNCTION pg_temp.expect_true(p_label text, p_sql text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE v boolean;
BEGIN
  EXECUTE 'SELECT (' || p_sql || ')' INTO v;
  INSERT INTO res VALUES (coalesce(v, false), p_label);
EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' [' || SQLSTATE || ']');
END $$;
