CREATE FUNCTION pg_temp.must_fail(p_label text, p_state text, p_sql text) RETURNS text LANGUAGE plpgsql AS $$
BEGIN
  EXECUTE p_sql;
  RETURN 'CAU TRUC KHONG DAT|' || p_label || ' (khong bi chan)';
EXCEPTION WHEN OTHERS THEN
  RETURN CASE WHEN SQLSTATE = p_state THEN 'CAU TRUC ok|' ELSE 'CAU TRUC KHONG DAT|' END || p_label || ' [' || SQLSTATE || ']';
END $$;
