-- expect_err: cau lenh PHAI loi dung SQLSTATE; p_token (neu co) phai la CONSTRAINT cua loi (ma loi co dinh cua trigger).
CREATE FUNCTION pg_temp.expect_err(p_label text, p_sql text, p_state text, p_token text DEFAULT NULL) RETURNS void LANGUAGE plpgsql AS $$
DECLARE v_c text;
BEGIN
  BEGIN
    EXECUTE p_sql;
    RAISE EXCEPTION 'q2_undo' USING ERRCODE = 'P0001';
  EXCEPTION
    WHEN raise_exception THEN INSERT INTO res VALUES (false, p_label || CASE WHEN SQLERRM = 'q2_undo' THEN ' [CHAY DUOC]' ELSE ' [' || SQLERRM || ']' END);
    WHEN OTHERS THEN
      GET STACKED DIAGNOSTICS v_c = CONSTRAINT_NAME;
      INSERT INTO res VALUES (SQLSTATE = p_state AND (p_token IS NULL OR v_c = p_token),
        p_label || CASE WHEN SQLSTATE = p_state AND (p_token IS NULL OR v_c = p_token) THEN '' ELSE ' [' || SQLSTATE || '/' || coalesce(v_c, '') || ' thay vi ' || p_state || '/' || coalesce(p_token, '') || ']' END);
  END;
END $$;
CREATE FUNCTION pg_temp.expect_true(p_label text, p_sql text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE v boolean;
BEGIN
  EXECUTE 'SELECT (' || p_sql || ')' INTO v;
  INSERT INTO res VALUES (coalesce(v, false), p_label);
EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' [' || SQLSTATE || ' ' || SQLERRM || ']');
END $$;
-- expect_run: cau lenh PHAI chay duoc, sau do bieu thuc kiem phai dung
CREATE FUNCTION pg_temp.expect_run(p_label text, p_sql text, p_check text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE v boolean;
BEGIN
  EXECUTE p_sql;
  EXECUTE 'SELECT (' || p_check || ')' INTO v;
  INSERT INTO res VALUES (coalesce(v, false), p_label);
EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' [' || SQLSTATE || ' ' || SQLERRM || ']');
END $$;

-- Dat tong dung theo cac dong hien co roi GUI
CREATE FUNCTION pg_temp.send_version(p_version uuid) RETURNS void LANGUAGE plpgsql AS $$
DECLARE b8 bigint; b10 bigint;
BEGIN
  SELECT coalesce(sum(amount_vnd) FILTER (WHERE vat_percent = 8), 0), coalesce(sum(amount_vnd) FILTER (WHERE vat_percent = 10), 0) INTO b8, b10 FROM public.quote_line WHERE quote_version_id = p_version;
  UPDATE public.quote_version SET base8_vnd = b8, base10_vnd = b10, subtotal_vnd = b8 + b10, vat8_vnd = (b8 * 8 + 50) / 100, vat10_vnd = (b10 * 10 + 50) / 100,
         total_vnd = b8 + b10 + (b8 * 8 + 50) / 100 + (b10 * 10 + 50) / 100 WHERE id = p_version;
  UPDATE public.quote_version SET status = 'SENT', quote_date = current_date, valid_until = current_date + 15, sent_by = '6b000000-0000-4000-8000-000000000021', sent_at = now() WHERE id = p_version;
END $$;
