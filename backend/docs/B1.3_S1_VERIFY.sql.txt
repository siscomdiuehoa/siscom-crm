-- Thu mot cau lenh: ok = true khi cau lenh BI TU CHOI voi ma loi mong doi (42501 insufficient_privilege / 42P01...), trong SAVEPOINT.
CREATE FUNCTION pg_temp.expect_denied(p_label text, p_sql text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_sql;
    RAISE EXCEPTION 's1_undo' USING ERRCODE = 'P0001';       -- chay duoc: huy moi hieu ung cua cau lenh vua chay, ghi KHONG DAT o handler
  EXCEPTION
    WHEN insufficient_privilege THEN INSERT INTO s1_result VALUES (p_label, true);
    WHEN raise_exception THEN INSERT INTO s1_result VALUES (p_label || CASE WHEN SQLERRM = 's1_undo' THEN ' [app CHAY DUOC]' ELSE ' [' || SQLERRM || ']' END, false);
    WHEN OTHERS THEN INSERT INTO s1_result VALUES (p_label || ' [loi khac: ' || SQLSTATE || ']', false);
  END;
END $$;

-- D12: GRANT boi role KHONG so huu va KHONG co grant option thi PostgreSQL CHI CANH BAO "no privileges were granted" va thoat 0
-- (khong RAISE 42501). Vi vay khong kiem bang ma loi, ma kiem HIEU QUA: sau khi thu GRANT, app van KHONG co quyen do (CI run
-- 41005714842 tren 8c773aa phat hien bai cu doi 42501 nen FAIL oan; kiem soat thuc te van dung).
CREATE FUNCTION pg_temp.expect_no_gain(p_label text, p_sql text, p_check text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE v_gained boolean;
BEGIN
  BEGIN
    EXECUTE p_sql;                       -- co the "thanh cong" kem WARNING, hoac bi tu choi: ca hai deu chap nhan
  EXCEPTION WHEN OTHERS THEN NULL;       -- bi tu choi cung tot; dieu quan trong la buoc kiem duoi
  END;
  EXECUTE 'SELECT ' || p_check INTO v_gained;   -- true = DA NANG DUOC QUYEN => KHONG DAT
  INSERT INTO s1_result VALUES (p_label, NOT coalesce(v_gained, true));
END $$;

SELECT pg_temp.expect_no_gain('D13 app GRANT ... WITH GRANT OPTION cho chinh minh => khong co hieu luc',
