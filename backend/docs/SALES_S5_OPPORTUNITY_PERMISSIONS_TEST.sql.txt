-- 3. SoD: admin (sysadmin) khong duoc cap opportunity.* ke ca qua API (trigger M1 tu choi nho phan loai BIZ; khoi EXCEPTION huy ca yeu cau/duyet vua tao)
CREATE FUNCTION pg_temp.sod_try(p_code text) RETURNS text LANGUAGE plpgsql AS $$
DECLARE v_req uuid;
BEGIN
  v_req := fn_auth_api_request_create('55402000-0000-4000-8000-000000000001', 'ROLE_PERMISSION_CHANGE', NULL, '55401000-0000-4000-8000-000000000001',
             jsonb_build_object('op', 'ADD', 'permissionId', (SELECT id FROM permission WHERE code = p_code), 'scope', 'ALL'), 60);
  PERFORM fn_auth_api_request_decide('55402000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('55402000-0000-4000-8000-000000000001', v_req);
  RETURN 'CHAY DUOC';
EXCEPTION WHEN OTHERS THEN RETURN SQLSTATE;
END $$;

CREATE FUNCTION pg_temp.cap(p_role uuid, p_code text, p_scope text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE v_req uuid;
BEGIN
  v_req := fn_auth_api_request_create('55402000-0000-4000-8000-000000000001', 'ROLE_PERMISSION_CHANGE', NULL, p_role,
             jsonb_build_object('op', 'ADD', 'permissionId', (SELECT id FROM permission WHERE code = p_code), 'scope', p_scope, 't', p_code), 60);
  PERFORM fn_auth_api_request_decide('55402000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('55402000-0000-4000-8000-000000000001', v_req);
END $$;
