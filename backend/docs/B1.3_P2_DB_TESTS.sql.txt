CREATE TEMP TABLE res (ok boolean, label text);
CREATE TEMP TABLE ids (name text PRIMARY KEY, id uuid);
-- Chay duoi quyen superuser (chu so huu)
CREATE FUNCTION pg_temp.expect_err(p_label text, p_stmt text, p_state text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_stmt;
    INSERT INTO res VALUES (false, p_label || ' - KHONG bi chan');
  EXCEPTION WHEN OTHERS THEN
    IF SQLSTATE = p_state THEN INSERT INTO res VALUES (true, p_label);
    ELSE INSERT INTO res VALUES (false, p_label || ' - SQLSTATE ' || SQLSTATE || ' (' || SQLERRM || '), can ' || p_state); END IF;
  END;
END $$;
CREATE FUNCTION pg_temp.expect_ok(p_label text, p_stmt text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN EXECUTE p_stmt; INSERT INTO res VALUES (true, p_label);
  EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')'); END;
END $$;
CREATE FUNCTION pg_temp.expect_true(p_label text, p_expr text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE r boolean;
BEGIN
  BEGIN EXECUTE 'SELECT (' || p_expr || ')' INTO r; INSERT INTO res VALUES (r IS TRUE, p_label);
  EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')'); END;
END $$;
-- Chay duoi vai p_role (SET LOCAL ROLE trong khoi con; loi => khoi con huy, vai tro tro ve superuser). Ghi ket qua SAU KHI RESET ROLE.
CREATE FUNCTION pg_temp.as_err(p_role text, p_label text, p_stmt text, p_state text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE format('SET LOCAL ROLE %I', p_role);
    EXECUTE p_stmt;
    EXECUTE 'RESET ROLE';
    INSERT INTO res VALUES (false, p_role || ': ' || p_label || ' - KHONG bi chan');
  EXCEPTION WHEN OTHERS THEN
    IF SQLSTATE = p_state THEN INSERT INTO res VALUES (true, p_role || ': ' || p_label);
    ELSE INSERT INTO res VALUES (false, p_role || ': ' || p_label || ' - SQLSTATE ' || SQLSTATE || ' (' || SQLERRM || '), can ' || p_state); END IF;
  END;
END $$;
CREATE FUNCTION pg_temp.as_ok(p_role text, p_label text, p_stmt text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE format('SET LOCAL ROLE %I', p_role);
    EXECUTE p_stmt;
    EXECUTE 'RESET ROLE';
    INSERT INTO res VALUES (true, p_role || ': ' || p_label);
  EXCEPTION WHEN OTHERS THEN
    INSERT INTO res VALUES (false, p_role || ': ' || p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')');
  END;
END $$;
CREATE FUNCTION pg_temp.as_true(p_role text, p_label text, p_expr text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE r boolean;
BEGIN
  BEGIN
    EXECUTE format('SET LOCAL ROLE %I', p_role);
    EXECUTE 'SELECT (' || p_expr || ')' INTO r;
    EXECUTE 'RESET ROLE';
    INSERT INTO res VALUES (r IS TRUE, p_role || ': ' || p_label);
  EXCEPTION WHEN OTHERS THEN
    INSERT INTO res VALUES (false, p_role || ': ' || p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')');
  END;
END $$;
-- Chay duoi vai p_role, luu ket qua uuid (hoac text) cua p_stmt (SELECT mot gia tri) vao ids(p_name)
CREATE FUNCTION pg_temp.as_cap(p_role text, p_label text, p_name text, p_stmt text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE r uuid;
BEGIN
  BEGIN
    EXECUTE format('SET LOCAL ROLE %I', p_role);
    EXECUTE p_stmt INTO r;
    EXECUTE 'RESET ROLE';
    INSERT INTO ids VALUES (p_name, r);
    INSERT INTO res VALUES (r IS NOT NULL, p_role || ': ' || p_label);
  EXCEPTION WHEN OTHERS THEN
    INSERT INTO res VALUES (false, p_role || ': ' || p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')');
  END;
END $$;
CREATE FUNCTION pg_temp.q(p_name text) RETURNS uuid LANGUAGE sql AS $$ SELECT id FROM ids WHERE name = p_name $$;
CREATE FUNCTION pg_temp.h(p_text text) RETURNS text LANGUAGE sql AS $$ SELECT encode(sha256(convert_to(p_text, 'UTF8')), 'hex') $$;
