SELECT current_database() AS db_rollback, current_user AS chay_bang;
BEGIN;
DO $$ BEGIN
  IF current_database() !~ '^siscom_crm_(restore|test)_[0-9]{8}_[0-9]{6}$'
     OR current_database() IN ('siscom_crm_restore_20261014_214105', 'siscom_crm_restore_20261015_214948') THEN
    RAISE EXCEPTION 'TU CHOI: % khong phai ban sao thu hop le cho rollback M1.2.', current_database();
  END IF;
  IF to_regprocedure('public.fn_cred_secret_guard()') IS NULL AND to_regprocedure('public.fn_auth_api_role_grant(uuid, uuid)') IS NULL THEN
    RAISE EXCEPTION 'M1.2 chua duoc ap. Khong co gi de rollback.';
  END IF;
  IF EXISTS (SELECT 1 FROM public.app_meta WHERE key = 'auth_bootstrap_done') THEN
    RAISE EXCEPTION 'TU CHOI: da co app_meta.auth_bootstrap_done (da khoi tao xac thuc).';
  END IF;
  IF (SELECT pg_get_userbyid(datdba) FROM pg_database WHERE datname = current_database()) <> 'siscom_app' THEN
    RAISE EXCEPTION 'TU CHOI: chu so huu database khong phai siscom_app (con dang tach role?). Chay B1.3_S1_ROLE_SPLIT_ROLLBACK.sql.txt truoc.';
  END IF;

CREATE TEMP VIEW v_m12_left AS SELECT
  (SELECT count(*) FROM pg_proc p JOIN pg_namespace n ON n.oid = p.pronamespace WHERE n.nspname = 'public'
     AND (p.proname LIKE 'fn\_auth\_api\_%' OR p.proname IN ('fn_auth_bootstrap_close', 'fn_cred_secret_guard', 'fn_cred_revoke_sessions'))) AS ham_con,
  (SELECT count(*) FROM pg_trigger WHERE NOT tgisinternal AND tgname IN ('trg_cred_secret_guard', 'trg_cred_revoke_sessions')) AS trigger_con,
  (SELECT count(*) FROM public._prisma_migrations WHERE migration_name = '20261020180000_b1_3_auth_m1_2_priv_api') AS lich_su_con;
SELECT * FROM v_m12_left;
DO $$ BEGIN
  IF (SELECT ham_con + trigger_con + lich_su_con FROM v_m12_left) <> 0 THEN
    RAISE EXCEPTION 'Rollback M1.2 chua sach. Huy transaction.';
  END IF;
END $$;
DROP VIEW v_m12_left;
COMMIT;
