CREATE TEMP TABLE res (ok boolean, label text);
CREATE FUNCTION pg_temp.expect_err(p_label text, p_stmt text, p_state text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN
    EXECUTE p_stmt;
    INSERT INTO res VALUES (false, p_label || ' - KHONG bi chan');
  EXCEPTION WHEN OTHERS THEN
    IF SQLSTATE = p_state THEN INSERT INTO res VALUES (true, p_label);
    ELSE INSERT INTO res VALUES (false, p_label || ' - SQLSTATE ' || SQLSTATE || ' (' || SQLERRM || '), can ' || p_state); END IF;
  END;
END $$;
CREATE FUNCTION pg_temp.expect_ok(p_label text, p_stmt text) RETURNS void LANGUAGE plpgsql AS $$
BEGIN
  BEGIN EXECUTE p_stmt; INSERT INTO res VALUES (true, p_label);
  EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')'); END;
END $$;
CREATE FUNCTION pg_temp.expect_true(p_label text, p_expr text) RETURNS void LANGUAGE plpgsql AS $$
DECLARE r boolean;
BEGIN
  BEGIN EXECUTE 'SELECT (' || p_expr || ')' INTO r; INSERT INTO res VALUES (r IS TRUE, p_label);
  EXCEPTION WHEN OTHERS THEN INSERT INTO res VALUES (false, p_label || ' - loi ' || SQLSTATE || ' (' || SQLERRM || ')'); END;
END $$;
CREATE FUNCTION pg_temp.pcr(p_id uuid, p_type text, p_emp uuid, p_role uuid, p_payload jsonb, p_by uuid, p_ttl interval DEFAULT interval '1 hour')
RETURNS void LANGUAGE sql AS $$
  SELECT set_config('app.actor_employee_id', p_by::text, true);
  INSERT INTO privileged_change_request (id, type, subject_employee_id, subject_role_id, payload, payload_hash, requested_by, expires_at)
  VALUES (p_id, p_type::privileged_change_type, p_emp, p_role, p_payload, encode(sha256(convert_to(p_payload::text, 'UTF8')), 'hex'), p_by, clock_timestamp() + p_ttl) $$;
CREATE FUNCTION pg_temp.ctx(p_req uuid, p_actor uuid) RETURNS void LANGUAGE sql AS $$
  SELECT set_config('app.priv_request_id', coalesce(p_req::text, ''), true), set_config('app.actor_employee_id', coalesce(p_actor::text, ''), true) $$;
CREATE FUNCTION pg_temp.approve(p_id uuid, p_by uuid) RETURNS void LANGUAGE sql AS $$
  SELECT set_config('app.actor_employee_id', p_by::text, true);
  UPDATE privileged_change_request SET status = 'APPROVED', approved_by = p_by WHERE id = p_id $$;
