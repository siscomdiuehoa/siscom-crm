END $$;

BEGIN
  v_req := fn_auth_api_request_create('f4a02000-0000-4000-8000-000000000001', 'ROLE_PERMISSION_CHANGE', NULL, 'f4a01000-0000-4000-8000-000000000001',
             jsonb_build_object('op', 'ADD', 'permissionId', (SELECT id FROM permission WHERE code = 'customer.export'), 'scope', 'ALL'), 60);
  PERFORM fn_auth_api_request_decide('f4a02000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('f4a02000-0000-4000-8000-000000000001', v_req);
  RETURN 'CHAY DUOC';

END $$;

BEGIN

  PERFORM fn_auth_api_request_decide('f4a02000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('f4a02000-0000-4000-8000-000000000001', v_req);
