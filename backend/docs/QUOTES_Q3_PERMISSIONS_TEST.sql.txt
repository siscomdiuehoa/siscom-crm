CREATE FUNCTION pg_temp.sod_try(p_code text) RETURNS text LANGUAGE plpgsql AS $$
DECLARE v_req uuid;
BEGIN
  v_req := fn_auth_api_request_create('6b302000-0000-4000-8000-000000000001', 'ROLE_PERMISSION_CHANGE', NULL, '6b301000-0000-4000-8000-000000000001',
             jsonb_build_object('op', 'ADD', 'permissionId', (SELECT id FROM permission WHERE code = p_code), 'scope', 'ALL'), 60);
  PERFORM fn_auth_api_request_decide('6b302000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('6b302000-0000-4000-8000-000000000001', v_req);
  RETURN 'CHAY DUOC';
EXCEPTION WHEN OTHERS THEN RETURN SQLSTATE;
END $$;

BEGIN

             jsonb_build_object('op', 'ADD', 'permissionId', (SELECT id FROM permission WHERE code = p_code), 'scope', p_scope, 't', p_code), 60);
  PERFORM fn_auth_api_request_decide('6b302000-0000-4000-8000-000000000002', v_req, true);
  PERFORM fn_auth_api_role_permission_change('6b302000-0000-4000-8000-000000000001', v_req);
END $$;
